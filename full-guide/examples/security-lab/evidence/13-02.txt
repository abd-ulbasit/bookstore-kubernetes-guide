## 4. What the provider identity can reach
$ kubectl auth can-i create namespaces --as=system:serviceaccount:p13-02-crossplane:provider-kubernetes
Warning: resource 'namespaces' is not namespace scoped

yes
$ kubectl auth can-i create networkpolicies -n p13-02-tenant --as=system:serviceaccount:p13-02-crossplane:provider-kubernetes
yes
$ kubectl auth can-i get secrets -n p13-02-tenant --as=system:serviceaccount:p13-02-crossplane:provider-kubernetes
no
$ kubectl auth can-i create clusterrolebindings --as=system:serviceaccount:p13-02-crossplane:provider-kubernetes
Warning: resource 'clusterrolebindings' is not namespace scoped in group 'rbac.authorization.k8s.io'

no
$ kubectl auth can-i '*' '*' --as=system:serviceaccount:p13-02-crossplane:provider-kubernetes
no
