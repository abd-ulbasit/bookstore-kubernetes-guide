## step 2: a runaway process reaches its memory limit (deliberate failure)
$ kubectl apply -f - <<'EOF'
apiVersion: v1
kind: ServiceAccount
metadata:
  name: demo-sa
  namespace: p01-03-bookstore
automountServiceAccountToken: false
---
apiVersion: v1
kind: Pod
metadata:
  name: memhog
  namespace: p01-03-bookstore
  labels:
    app: memhog
spec:
  serviceAccountName: demo-sa
  automountServiceAccountToken: false
  restartPolicy: Never              # keep the terminated state to read
  securityContext:
    runAsNonRoot: true
    runAsUser: 65534
    runAsGroup: 65534
    seccompProfile:
      type: RuntimeDefault
  containers:
    - name: hog                     # DELIBERATE: writes past its memory limit
      image: busybox@sha256:73aaf090f3d85aa34ee199857f03fa3a95c8ede2ffd4cc2cdb5b94e566b11662
      command: ["sh", "-c", "dd if=/dev/zero of=/scratch/fill bs=1M count=400; echo written; sleep 3600"]
      securityContext:
        allowPrivilegeEscalation: false
        readOnlyRootFilesystem: true
        capabilities:
          drop: ["ALL"]
      resources:
        requests: { cpu: 50m, memory: 64Mi }
        limits: { cpu: 250m, memory: 128Mi }
      volumeMounts:
        - name: scratch
          mountPath: /scratch
  volumes:
    - name: scratch
      emptyDir:
        medium: Memory              # tmpfs: its files count against the memory limit
---
apiVersion: v1
kind: Pod
metadata:
  name: cpu-fill
  namespace: p01-03-bookstore
  labels:
    app: cpu-fill
spec:
  serviceAccountName: demo-sa
  automountServiceAccountToken: false
  securityContext:
    runAsNonRoot: true
    runAsUser: 65534
    runAsGroup: 65534
    seccompProfile:
      type: RuntimeDefault
  containers:
    - name: fill
      image: busybox@sha256:73aaf090f3d85aa34ee199857f03fa3a95c8ede2ffd4cc2cdb5b94e566b11662
      command: ["sleep", "3600"]
      securityContext:
        allowPrivilegeEscalation: false
        readOnlyRootFilesystem: true
        capabilities:
          drop: ["ALL"]
      resources:
        requests: { cpu: 1500m, memory: 64Mi }
        limits: { cpu: 1500m, memory: 128Mi }
EOF

## step 3: admission refuses what the namespace cannot hold (deliberate failure)
$ kubectl apply --dry-run=server -f - <<'EOF'
apiVersion: v1
kind: Pod
metadata:
  name: too-big
  namespace: p01-03-bookstore
spec:
  serviceAccountName: demo-sa
  automountServiceAccountToken: false
  securityContext:
    runAsNonRoot: true
    runAsUser: 65534
    runAsGroup: 65534
    seccompProfile:
      type: RuntimeDefault
  containers:
    - name: c                       # DELIBERATE: above the 1Gi max
      image: busybox@sha256:73aaf090f3d85aa34ee199857f03fa3a95c8ede2ffd4cc2cdb5b94e566b11662
      command: ["sleep", "3600"]
      securityContext:
        allowPrivilegeEscalation: false
        readOnlyRootFilesystem: true
        capabilities:
          drop: ["ALL"]
      resources:
        requests: { cpu: 50m, memory: 1536Mi }
        limits: { cpu: 250m, memory: 1536Mi }

  name: too-much
