# Transcript for 05-security/05-kernel-isolation.md, lab steps 1 to 7.
# Mode: P05_RUNTIME=0 (the seccomp steps, and read-only checks of AppArmor and
# the runtime). Steps 5 and 6 changed nothing on the node.
# Tier A: kind on a Linux host (kernel 7.0.0, AppArmor enabled in the kernel).
# Cluster: the shared sf-pilot kind cluster, node Kubernetes v1.37.0, kubectl
# v1.37.2. Node containerd 2.3.4. Not run on macOS.
# Recorded by running labs/05-05.sh from examples/security-lab/ on the lab host.
#
# What did not work the first time:
# - Audit (step 4a): the kernel printed 20 audit records and dropped 401 more,
#   so none of the busybox workload's records reached the journal. Without an
#   audit daemon, these records go through printk, which the kernel rate-limits.
#   An allowlist could not be derived from the log, so the enforce profile is a
#   deny-list that refuses personality only.
# - AppArmor (step 5) and gVisor (step 6) did not run on a throwaway cluster. A
#   fresh kind node on the lab host stalled in kubeadm init, on six attempts,
#   with and without the securityfs mount. This run is on the shared cluster,
#   where neither can be installed without restarting the node's containerd.
# - busybox has no /etc/os-release. The workload reads /etc/passwd instead.
#
## node: sf-pilot-control-plane, kubectl v1.37.2

## step 1: the node shares the lab host kernel, and sees AppArmor
$ docker exec sf-pilot-control-plane uname -r
docker exec sf-pilot-control-plane cat /sys/module/apparmor/parameters/enabled
docker exec sf-pilot-control-plane sh -c 'test -d /sys/kernel/security/apparmor && echo apparmor-securityfs-present || echo apparmor-securityfs-absent'
docker exec sf-pilot-control-plane sh -c 'test -e /sys/fs/selinux && echo selinux-present || echo selinux-absent'

## step 2: two namespaces, one that permits everything and one restricted
$ kubectl create ns p05-05-attack
kubectl label ns p05-05-attack pod-security.kubernetes.io/enforce=privileged
kubectl create ns p05-05-restricted
kubectl label ns p05-05-restricted \
  pod-security.kubernetes.io/enforce=restricted \
  pod-security.kubernetes.io/audit=restricted \
  pod-security.kubernetes.io/warn=restricted
namespace/p05-05-attack created
namespace/p05-05-attack labeled
namespace/p05-05-restricted created
namespace/p05-05-restricted labeled

## step 3: seccomp Unconfined against RuntimeDefault, for one personality call
$ kubectl apply -f unconfined.yaml
kubectl apply -f runtimedefault.yaml
kubectl -n p05-05-attack wait --for=condition=Ready pod/p05-unconfined --timeout=180s
kubectl -n p05-05-restricted wait --for=condition=Ready pod/p05-runtimedefault --timeout=180s
pod/p05-unconfined created

pod/p05-runtimedefault condition met

## step 3a: the Unconfined Pod: no filter, the personality call succeeds
$ kubectl -n p05-05-attack exec p05-unconfined -- sh -c 'grep "^Seccomp:" /proc/self/status | tr "\t" " "; setarch linux64 -R /bin/true 2>&1; echo exit=$?'
Seccomp: 0
exit=0

## step 3b: the RuntimeDefault Pod: the same call is refused by the filter
$ kubectl -n p05-05-restricted exec p05-runtimedefault -- sh -c 'grep "^Seccomp:" /proc/self/status | tr "\t" " "; setarch linux64 -R /bin/true 2>&1; echo exit=$?'
Seccomp: 2
linux64: personality(0x40000): Operation not permitted
exit=1

## step 3c: restricted refuses the Unconfined spec at admission (server dry run)
$ kubectl apply --dry-run=server -f unconfined-restricted.yaml
Error from server (Forbidden): error when creating "unconfined-restricted.yaml": pods "p05-unconfined-restricted" is forbidden: violates PodSecurity "restricted:latest": seccompProfile (pod must not set securityContext.seccompProfile.type to "Unconfined")

## step 4: audit, then enforce. A custom Localhost profile, logged first

$ kubectl -n p05-05-restricted exec p05-audit -- sh -c 'id -u; cat /etc/passwd >/dev/null; ls / >/dev/null; echo done'

## step 4a: the kernel log: the records it printed, and the ones it dropped

## step 4b: enforce profile: allow everything except personality, which returns EPERM
$ docker exec sf-pilot-control-plane cat /var/lib/kubelet/seccomp/profiles/p05-deny-personality.json
{
  "defaultAction": "SCMP_ACT_ALLOW",
  "architectures": ["SCMP_ARCH_X86_64"],
  "syscalls": [
    {
      "names": ["personality"],
      "action": "SCMP_ACT_ERRNO",
      "errnoRet": 1
    }
  ]
}
