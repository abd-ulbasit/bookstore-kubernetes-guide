## step 2: two namespaces, one that permits everything and one restricted
$ kubectl create ns p05-05-attack
kubectl label ns p05-05-attack pod-security.kubernetes.io/enforce=privileged
kubectl create ns p05-05-restricted
kubectl label ns p05-05-restricted \
  pod-security.kubernetes.io/enforce=restricted \
  pod-security.kubernetes.io/audit=restricted \
  pod-security.kubernetes.io/warn=restricted
namespace/p05-05-attack created
namespace/p05-05-attack labeled
namespace/p05-05-restricted created
namespace/p05-05-restricted labeled

## step 3: seccomp Unconfined against RuntimeDefault, for one personality call
$ kubectl apply -f unconfined.yaml
kubectl apply -f runtimedefault.yaml
kubectl -n p05-05-attack wait --for=condition=Ready pod/unconfined --timeout=180s
kubectl -n p05-05-restricted wait --for=condition=Ready pod/runtimedefault --timeout=180s
pod/unconfined created
pod/runtimedefault created
pod/unconfined condition met
pod/runtimedefault condition met

## step 3a: the Unconfined Pod: no filter, the personality call succeeds
$ kubectl -n p05-05-attack exec unconfined -- sh -c 'grep "^Seccomp:" /proc/self/status | tr "\t" " "; setarch linux64 -R /bin/true 2>&1; echo exit=$?'
Seccomp: 0
exit=0

## step 3b: the RuntimeDefault Pod: the same call is refused by the filter
$ kubectl -n p05-05-restricted exec runtimedefault -- sh -c 'grep "^Seccomp:" /proc/self/status | tr "\t" " "; setarch linux64 -R /bin/true 2>&1; echo exit=$?'
Seccomp: 2
linux64: personality(0x40000): Operation not permitted
exit=1

## step 3c: restricted refuses the Unconfined spec at admission (server dry run)
$ kubectl apply --dry-run=server -f unconfined-restricted.yaml
Error from server (Forbidden): error when creating "unconfined-restricted.yaml": pods "unconfined-restricted" is forbidden: violates PodSecurity "restricted:latest": seccompProfile (pod must not set securityContext.seccompProfile.type to "Unconfined")
