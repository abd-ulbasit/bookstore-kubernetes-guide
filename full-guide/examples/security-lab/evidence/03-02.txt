## step 2: the 03.01 Secret, then a second Secret with a value generated now
$ sed -e 's/namespace: bookstore/namespace: p03-02-secrets/' examples/bookstore/raw-manifests/16-db-credentials.yaml | kubectl apply -f -   # 03.01's Secret, demo literal
secret/db-credentials created
$ kubectl get secret db-credentials -n p03-02-secrets -o name
secret/db-credentials
$ PW=$(openssl rand -hex 16)   # generated now, never printed
$ sed -e 's/devpassword/$PW/' -e 's/name: db-credentials/name: db-credentials-generated/' -e 's/namespace: bookstore/namespace: p03-02-secrets/' examples/bookstore/raw-manifests/16-db-credentials.yaml | kubectl apply -f -
secret/db-credentials-generated created
$ kubectl get secret db-credentials-generated -n p03-02-secrets -o jsonpath='{.data.POSTGRES_DB}'; echo
Ym9va3N0b3Jl
$ kubectl get secret db-credentials-generated -n p03-02-secrets -o jsonpath='{.data.POSTGRES_DB}' | base64 -d; echo
bookstore
$ kubectl get secret db-credentials-generated -n p03-02-secrets -o jsonpath='{.data.POSTGRES_PASSWORD}' | base64 -d | wc -c | tr -d ' '
32
$ decoded POSTGRES_PASSWORD equals $PW (compared, not printed)
yes
$ kubectl get secret db-credentials-generated -n p03-02-secrets -o jsonpath='{.metadata.annotations.kubectl\.kubernetes\.io/last-applied-configuration}' | grep -c stringData
1
$ kubectl get secret db-credentials-generated -n p03-02-secrets -o jsonpath='{...last-applied-configuration}' | grep -c "$PW"   # the generated value, counted not printed
1

## step 6: the Secret as files, the mode an inspection Pod sees, and exec
$ kubectl apply -f -   # secret-files: runs as secret-reader, mounts db-credentials-generated with defaultMode 0440 and fsGroup 65532, then sleeps
pod/secret-files created
$ kubectl wait pod/secret-files -n p03-02-secrets --for=condition=Ready --timeout=300s
pod/secret-files condition met
$ kubectl logs secret-files -n p03-02-secrets
uid=65532 gid=65532 groups=65532
total 12
-r--r-----    1 0        65532            9 Oct 31 02:27 POSTGRES_DB
-r--r-----    1 0        65532           32 Oct 31 02:27 POSTGRES_PASSWORD
-r--r-----    1 0        65532            9 Oct 31 02:27 POSTGRES_USER
tmpfs /etc/creds tmpfs ro,relatime,size=65536k,inode64,noswap 0 0
$ kubectl exec secret-files -n p03-02-secrets -- sh -c 'wc -c < /etc/creds/POSTGRES_PASSWORD'

$ kubectl delete pod secret-files -n p03-02-secrets --now
pod "secret-files" deleted from p03-02-secrets namespace

## step 7: an immutable Secret refuses an in-place change
$ kubectl apply -f -   # immutable-demo: Secret with immutable: true, value generated now
secret/immutable-demo created
$ kubectl get secret immutable-demo -n p03-02-secrets -o jsonpath='{.immutable}'; echo
true
$ kubectl patch secret immutable-demo -n p03-02-secrets --type merge -p "{\"stringData\":{\"token\":\"$(openssl rand -hex 16)\"}}"
The Secret "immutable-demo" is invalid: data: Forbidden: field is immutable when `immutable` is set
$ kubectl delete secret immutable-demo -n p03-02-secrets
secret "immutable-demo" deleted from p03-02-secrets namespace
