## step 1: a restricted namespace and a default-deny NetworkPolicy
$ kubectl apply -f - <<'EOF'   # Namespace bookstore-ml and default-deny NetworkPolicy
namespace/p12-01-bookstore-ml created
networkpolicy.networking.k8s.io/default-deny created

## step 2: the restricted labels, and the policy stored in the namespace
$ kubectl get namespace p12-01-bookstore-ml -o jsonpath='{.metadata.labels}' | tr ',' '\n' | grep pod-security

"pod-security.kubernetes.io/audit-version":"latest"
"pod-security.kubernetes.io/enforce":"restricted"
"pod-security.kubernetes.io/enforce-version":"latest"
"pod-security.kubernetes.io/warn":"restricted"
"pod-security.kubernetes.io/warn-version":"latest"}

## step 3: a deliberate refusal, a Pod with no securityContext

exit status: 1
